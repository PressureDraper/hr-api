MÓDULO INCIDENCIAS
	***ROL BASE QUE TODOS LOS INVOLUCRADOS TIENEN QUE TENER***
	rol: rhumanos_incidencias
	permission: rhumanos_incidences_ver
	
	***PARA SOLO VER EL MODULO DE INCIDENCIAS SIN PERMISOS NI ACCIONES***
	rol: rhumanos_incidencias_visitante
	permission: 
			rhumanos_incidences_ver
		    rhumanos_kardex_ver
		    rhumanos_vacaciones_ver
		    rhumanos_festivos_ver
	
	-KARDEX:
		***PARA JEFATURAS CONSULTA EXTERNA QUE SOLO PUEDEN VER EL KARDEX***
		rol: 
		rhumanos_kardex
		
		permission:
		rhumanos_incidences_ver
		rhumanos_kardex_ver
		
		***PARA RECURSOS HUMANOS***
		rol:
		rhumanos_control_asistencial
		
		permission:
		rhumanos_kardex_ver = control asistencial tab
		rhumanos_kardex_editar = control asistencial acciones
		rhumanos_kardex_vista_anual = habilita funcion para navegar en el calendario por año
		
		rol:
		rhumanos_reporte_checadas = genera el reporte de checadas en el modulo de reportes
		
		permission:
		reportes_rhumanos_plantilla
		reportes_rhumanos_checadas

		***PARA CASOS ESPECIALES QUE PUEDEN VER EL CALENDARIO POR AÑO***
		rol: 
		rhumanos_kardex_vista_anual
		
		permission:
		rhumanos_kardex_vista_anual = habilita funcion para navegar en el calendario por año
		
	-PERMISOS:
		rol:
		rhumanos_permisos
		
		permission:
		rhumanos_permisos_ver = permisos tab
		rhumanos_permisos_eliminar = boton eliminar en el calendario al seleccionar un permiso
		//no es necesario el rol de editar
	
	-VACACIONES:
		rol:
		rhumanos_vacaciones
		
		permission:
		rhumanos_vacaciones_ver = vacaciones tab
		rhumanos_vacaciones_editar = vacaciones acciones
	
	-FESTIVOS:
		rol:
		rhumanos_festivos
		
		permission:
		rhumanos_festivos_ver = festivos tab
		rhumanos_festivos_editar = festivos acciones
	
MÓDULO FIRMAS
	***ROL BASE QUE TODOS LOS INVOLUCRADOS TIENEN QUE TENER***
	rol: rhumanos_firmas
	permission: rhumanos_firmas_ver

	***PARA AQUELLOS QUE FIRMAN (FIRMAN SOLO UNA VEZ Y NO PUEDEN VER LAS FIRMAS DE LOS DEMÁS)***
	rol: rhumanos_firmas_agregar
	permission: 
	rhumanos_firmas_eliminar
	rhumanos_firmas_editar
	rhumanos_firmas_ver_unico
